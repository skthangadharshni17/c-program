#include<stdio.h>
void swap(int*,int*);
int main()
{
   int x,y;
   printf("Enter the values of x and y:");
   scanf("%d%d",&x,&y);
   printf("Before swapping x=%d,y=%d",x,y);
   swap(&x,&y);
   printf("\nAfter swapping x=%d,y=%d",x,y);
}
void swap(int*p1,int*p2)
{
   int temp;
   temp=*p1;
   *p1=*p2;
   *p2=temp;
}
