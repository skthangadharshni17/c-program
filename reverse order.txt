#include<stdio.h>
#include<stdlib.h>
int main()
{
   int n,*ptr,i;
   printf("Enter the number of terms:");
   scanf("%d",&n);
   ptr=(int*)calloc(n,sizeof(int));
   printf("Enter the elements in array:");
   for(i=0;i<n;i++)
   {
      scanf("%d",ptr+i);
   }
   printf("The elements in reversed order:");
   for(i=n-1;i>=0;i--)
   {
      printf("%d\t",*(ptr+i));
   }
   free(ptr);
}
