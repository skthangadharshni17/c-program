#include<stdio.h>
#include<string.h>
int main()
{
   int i,j,l,n;
   int count=0;
   char str[100],letter;
   int menu;
   printf("Enter the word:");
   scanf("%s",&str);
   printf("Enter the menu:");
   scanf("%d",&menu);
   l=0;
   i=0;
   while(str[i]!='\0')
   {
      l=l+1;
      i=i+1;
   }
   n=l-1;
   switch(menu)
   {
      case 1:
          printf("Enter the letter:");
          scanf("%s",&letter);
         for(i=0;i<l;i++)
         {
            if(str[i]==letter)
            {
               printf("The letter is in %d position",i);
            }
         }
         break;
      case 2:
         printf("Enter the position:");
         scanf("%d",&i);
         printf("The letter in %d position is %c",i,str[i]);
         break;
      case 3:
          printf("Enter the letter:");
          scanf("%s",&letter);
         j=0;
         for(i=0;i<l;i++)
         {
            if(str[i]==letter)
            {
               j++;
            }
         }
            printf("The occurance=%d",j);
          break;
      case 4:
          for(i=0,j=n;j>=0;i++,j--)
          {
             if(str[i]==str[j])
                count++;
          }
          if(count==l)
          {
             printf("Palindrome");
          }
          else
          {
             printf("Not palindrome");
          }
   }
}
