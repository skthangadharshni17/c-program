#include<stdio.h>
#include<stdlib.h>
int main()
{
   int n,j=0,k=0,i;
   int *p1,*po,*pe,t,s;
   printf("Enter the number of students:");
   scanf("%d",&n);
   p1=(int*)malloc(n*sizeof(int));
   printf("Enter the first roll number:");
   scanf("%d",&s);
   t=s;
   for(i=0;i<n;i++)
   {

      p1[i]=t;
      t++;
   }
   if(n%2==0)
   {
      po=(int*)malloc((n/2)*sizeof(int));
      pe=(int*)malloc((n/2)*sizeof(int));
   }
   else
   {
         po=(int*)malloc(((n/2)+1)*sizeof(int));
         pe=(int*)malloc((n/2)*sizeof(int));
      }
      for(i=0;i<=n;i++)
      {
         if(s%2==0)
         {
            if(i%2==0)
            {
               pe[j]=p1[i];
               j++;
            }
            else
            {
               po[k]=p1[i];
               k++;
            }
         }
         else

         {
            if(i%2==0)
            {
               pe[j]=p1[i];
               j++;
            }
            else
            {
               po[k]=p1[i];
               k++;
            }
         }
      }
      printf("Enter the students under staff advisor 1:");
      for(i=0;i<j;i++)
      {
         printf("%d\n",po[i]);
      }
      printf("Enter the students under staff advisor 2:");
      for(i=0;i<k;i++)
      {
         printf("%d\n",pe[i]);
      }
   }
