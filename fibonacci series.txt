#include<stdio.h>
#include<stdlib.h>
int main()
{
   int n,*ptr,i;
   printf("Enter the number of terms:");
   scanf("%d",&n);
   ptr=(int*)calloc(n,sizeof(int));
   *(ptr+1)=1;
   for(i=2;i<=n;i++)
   {

      *(ptr+i)=*(ptr+i-1)+*(ptr+i-2);
   }
   printf("The fibonacci series is");
   for(i=0;i<=n-1;i++)
   {
      printf("%d",*(ptr+i));
   }
}
